        // 測試 5: 狀態訂閱
        window.testSubscribe = function() {
            if (window.unsubscribe) {
                window.log('subscribeLog', '⚠️ 已在監聽中', true);
                return;
            }

            const logElement = document.getElementById('subscribeLog');
            window.unsubscribe = window.state.subscribe((newState) => {
                const timestamp = new Date().toLocaleTimeString('zh-TW');
                const log = `[${timestamp}] 狀態已更新\n人員數: ${newState.people.length}\n`;
                logElement.textContent = log + logElement.textContent;
            });

            window.log('subscribeLog', '✅ 監聽已啟動');
        };

        window.testUnsubscribe = function() {
            if (window.unsubscribe) {
                window.unsubscribe();
                window.unsubscribe = null;
                window.log('subscribeLog', '✅ 監聽已停止');
            } else {
                window.log('subscribeLog', '⚠️ 尚未啟動監聽', true);
            }
        };

        window.triggerStateChange = function() {
            const randomName = '測試_' + Math.floor(Math.random() * 1000);
            window.state.addPerson(randomName);
            window.updateDisplay();
        };

        // 初始化顯示
        window.updateDisplay();
        console.log('✅ 模組載入成功！');
